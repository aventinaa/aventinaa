    <script>
  // 1. Scroll-Reveal Animation Logic
  const observerOptions = {
    root: null,
    rootMargin: '0px',
    threshold: 0.15
  };

  const revealObserver = new IntersectionObserver((entries, observer) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        entry.target.classList.add('visible');
        observer.unobserve(entry.target); // Animate only once
      }
    });
  }, observerOptions);

  document.querySelectorAll('.reveal').forEach(el => {
    revealObserver.observe(el);
  });

  // 2. Dynamic Text File Loader (Link1.txt, Link2.txt, etc.)
  async function loadTextFiles() {
    const textElements = document.querySelectorAll('[data-text-file]');
    
    textElements.forEach(async (el) => {
      const fileName = el.getAttribute('data-text-file');
      try {
        const response = await fetch(fileName);
        if (response.ok) {
          const text = await response.text();
          el.innerText = text.trim();
        }
      } catch (err) {
        console.warn(`Could not load ${fileName}:`, err);
      }
    });
  }

  // 3. Auto-Detect Extra Images (Image9, Image10, etc.)
  async function autoDetectImages() {
    const galleryContainer = document.getElementById('dynamic-gallery');
    if (!galleryContainer) return;

    let imageIndex = 9; // Starting check from Image9
    let consecutiveFailures = 0;
    const maxFailures = 2; // Stop checking after 2 missing images in a row

    while (consecutiveFailures < maxFailures) {
      const extensions = ['jpg', 'jpeg', 'png', 'webp'];
      let found = false;

      for (const ext of extensions) {
        const imagePath = `images/Image${imageIndex}.${ext}`;
        const exists = await checkImageExists(imagePath);

        if (exists) {
          appendImageToGallery(galleryContainer, imagePath, imageIndex);
          found = true;
          consecutiveFailures = 0;
          break;
        }
      }

      if (!found) {
        consecutiveFailures++;
      }

      imageIndex++;
    }
  }

  // Helper: Check if image path returns 200 OK
  function checkImageExists(url) {
    return new Promise((resolve) => {
      const img = new Image();
      img.onload = () => resolve(true);
      img.onerror = () => resolve(false);
      img.src = url;
    });
  }

  // Helper: Inject new image node into DOM with reveal effect
  function appendImageToGallery(container, src, index) {
    const card = document.createElement('div');
    card.className = 'gallery-card reveal';
    card.innerHTML = `
      <img src="${src}" alt="Aventinaa Collection item ${index}" loading="lazy">
    `;
    container.appendChild(card);
    revealObserver.observe(card); // Attach reveal animation to new element
  }

  // Initialize on page load
  window.addEventListener('DOMContentLoaded', () => {
    loadTextFiles();
    autoDetectImages();
  });
</script>
